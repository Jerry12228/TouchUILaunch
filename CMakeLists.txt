cmake_minimum_required(VERSION 3.24)
project(ZZZTouchUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /EHsc /utf-8)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ZZZTouchUI SHARED payload.cpp)
target_link_libraries(ZZZTouchUI PRIVATE user32 bcrypt)
add_executable(ZZZTouchLauncher launcher.cpp)
target_link_libraries(ZZZTouchLauncher PRIVATE user32 bcrypt shell32 advapi32 ole32)
add_executable(TouchStateTests touch_state_tests.cpp)
add_executable(TouchBridgeTests bridge_tests.cpp)
target_link_libraries(TouchBridgeTests PRIVATE user32 bcrypt)
add_executable(LauncherElevationTests elevation_tests.cpp)
target_link_libraries(LauncherElevationTests PRIVATE shell32 advapi32 ole32)
add_executable(ProfileDiscoveryTests discovery_tests.cpp)
target_link_libraries(ProfileDiscoveryTests PRIVATE bcrypt)
enable_testing()
add_test(NAME TouchState COMMAND TouchStateTests)
add_test(NAME TouchBridge31 COMMAND TouchBridgeTests 3.1)
add_test(NAME TouchBridge32 COMMAND TouchBridgeTests 3.2)
add_test(NAME LauncherElevation COMMAND LauncherElevationTests)
add_test(NAME ProfileDiscovery31 COMMAND ProfileDiscoveryTests "${CMAKE_CURRENT_SOURCE_DIR}/../Client/ZenlessZoneZero Game/GameAssembly.dll")
add_test(NAME ProfileDiscovery32 COMMAND ProfileDiscoveryTests "${CMAKE_CURRENT_SOURCE_DIR}/../Client/3.2/GameAssembly.dll")
